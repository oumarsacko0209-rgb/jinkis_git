
pipeline {
    agent any

    options {
        timestamps()
    }

    stages {
        stage('Cloner le depot') {
            steps {
                checkout scm
            }
        }

        stage('Verification du depot') {
            steps {
                bat 'echo === Verification du depot ==='
                bat 'git status'
            }
        }

        stage('Afficher le contenu du projet') {
            steps {
                bat 'echo === Contenu du projet ==='
                bat 'dir'
            }
        }

        stage('Simulation du deploiement') {
            steps {
                bat 'echo Le fichier index.html est pret a etre affiche'
            }
        }

        stage('Fin du build') {
            steps {
                bat 'echo SUCCESS'
            }
        }
    }

    post {
        success {
            echo 'Build OK'
        }
        failure {
            echo 'Build KO'
        }
    }
}
